cmake_minimum_required(VERSION 3.22)
project(hermes_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(whisper GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git GIT_TAG 0c7be696371069e0baa2515ed7aa08d5a5473aa6)
FetchContent_MakeAvailable(whisper)
add_library(hermes_whisper SHARED whisper_jni.cpp)
target_link_libraries(hermes_whisper PRIVATE whisper android log)
